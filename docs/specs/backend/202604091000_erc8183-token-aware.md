## 一、背景与目标

`BudgetSet` 事件携带 `paymentToken` 地址和 `amount`，但当前实现未捕获这些信息。改造目标：

1. 捕捉每个 job 的 `payment_token`/`token_symbol`/`budget_usd` 并固化到 `commerce_actions`
2. 为 stats 统计提供跨 token 可比的 USD 加权能力
3. 为后续真实花费记录（第二阶段）预留字段

## 四、Indexer 实现

### 4.1 Token 信息缓存（`indexer/processor/token.go`）

#### 4.1.1 Token Info 缓存（symbol + decimals 同时获取）

```go
type tokenInfoCache struct {
    mu     sync.RWMutex
    tokens map[string]TokenInfo  // token address → {symbol, decimals}
}

type TokenInfo struct {
    Symbol   string
    Decimals uint8
}

var globalTokenInfoCache tokenInfoCache

// ResolveTokenInfo: 先查缓存，未命中则串行调用 symbol() + decimals()
func ResolveTokenInfo(ctx context.Context, client *ethclient.Client, addr string) (*TokenInfo, error) {
    globalTokenInfoCache.mu.RLock()
    if info, ok := globalTokenInfoCache.tokens[addr]; ok {
        globalTokenInfoCache.mu.RUnlock()
        return &info, nil
    }
    globalTokenInfoCache.mu.RUnlock()

    // 串行调用 symbol() + decimals()，同时获取
    info, err := fetchTokenInfo(ctx, client, common.HexToAddress(addr))
    if err != nil {
        return &TokenInfo{Symbol: "???", Decimals: 18}, err  // fallback
    }

    globalTokenInfoCache.mu.Lock()
    globalTokenInfoCache.tokens[addr] = *info
    globalTokenInfoCache.mu.Unlock()
    return info, nil
}

// fetchTokenInfo: 串行调用 symbol() + decimals()，1 次 RPC（缓存命中时为 0）
func fetchTokenInfo(ctx context.Context, client *ethclient.Client, addr common.Address) (*TokenInfo, error) {
    symbol, err := callSymbol(ctx, client, addr)
    if err != nil {
        return &TokenInfo{Symbol: "???", Decimals: 18}, err
    }

    decimals, err := callDecimals(ctx, client, addr)
    if err != nil {
        return &TokenInfo{Symbol: symbol, Decimals: 18}, err  // fallback decimals
    }

    return &TokenInfo{Symbol: symbol, Decimals: decimals}, nil
}
```

> 注：缓存命中时 0 次 RPC；miss 时 2 次 RPC（symbol + decimals 串行）

#### 4.1.2 Price 缓存（纯懒加载）

```go
type tokenPriceEntry struct {
    price     float64
    fetchedAt time.Time
}

var globalPriceCache struct {
    mu sync.RWMutex
    m  map[string]tokenPriceEntry  // token address → price
}

const priceCacheTTL = 1 * time.Hour

// GetHistoricalUSDBudget: 返回 budget * price（快照）
// 纯懒加载：miss 时同步调用 historical price API
// decimals 参数用于正确换算 token 精度
// blockTimestamp 使用事件中的 block_timestamp（秒级 Unix 时间戳）
func GetHistoricalUSDBudget(ctx context.Context, platform, tokenAddr string, amount *big.Int, decimals uint8, blockTimestamp uint64) (float64, error) {
    // blockTimestamp (秒) → DD-MM-YYYY 格式
    dateStr := time.Unix(int64(blockTimestamp), 0).UTC().Format("02-01-2006")

    globalPriceCache.mu.RLock()
    cached, ok := globalPriceCache.m[tokenAddr]
    globalPriceCache.mu.RUnlock()

    var price float64
    if ok && time.Since(cached.fetchedAt) < priceCacheTTL {
        price = cached.price
    } else {
        var err error
        price, err = fetchHistoricalPrice(ctx, platform, tokenAddr, dateStr)
        if err != nil {
            log.Warnf("fetchHistoricalPrice failed for %s, using default 1: %v", tokenAddr, err)
            price = 1.0
        }
        globalPriceCache.mu.Lock()
        globalPriceCache.m[tokenAddr] = tokenPriceEntry{price: price, fetchedAt: time.Now()}
        globalPriceCache.mu.Unlock()
    }

    return float64(amount.Int64()) * price / math.Pow10(int(decimals)), nil
}
```

**策略**：不做主动预热，采用纯懒加载
- 每次 `GetHistoricalUSDBudget` / `GetCurrentUSDBudget` 调用时，先查缓存
- Cache miss 时调用 API，结果写入缓存供后续使用
- 无需预先知道所有 token 地址

#### 4.1.3 Job Budget 快照缓存

```go
var jobBudgetCache sync.Map  // map[uint64]*jobBudgetEntry

type jobBudgetEntry struct {
    PaymentToken string
    TokenSymbol  string
    Budget       *big.Int
    BudgetUSD    float64
}

// SetJobBudgetCache: handleBudgetSet 时写入
func SetJobBudgetCache(jobID uint64, entry *jobBudgetEntry) {
    jobBudgetCache.Store(jobID, entry)
}

// GetJobBudgetFromCache: 后续事件（funded/submitted/终态）直接取快照
func GetJobBudgetFromCache(jobID uint64) (*jobBudgetEntry, bool) {
    val, ok := jobBudgetCache.Load(jobID)
    if !ok {
        return nil, false
    }
    return val.(*jobBudgetEntry), true
}
```

### 4.2 `handleBudgetSet` 改动

**输入**：BudgetSet event (jobId, token, amount)

**处理**：
1. `paymentToken` = `ev.Token.String()`（事件直接取，零额外 RPC）
2. `tokenInfo` = `ResolveTokenInfo()`（首次调用写缓存，同时拿 symbol + decimals）
3. `budgetUSD` = `GetHistoricalUSDBudget()`（快照）
4. `budget` = `bigIntToFloat(ev.Amount)`
5. 查询 `job`（已有 getJobWithRetry）获取 `client` 地址
6. 写入 `commerce_actions`（client role）
7. **写入 `jobBudgetCache`**（供后续事件复用）

```go
func (p *CommerceProcessor) handleBudgetSet(e types.Log) error {
    ev, err := p.commerceContract.ParseBudgetSet(e)
    // ...
    paymentToken := ev.Token.String()
    tokenInfo, _ := ResolveTokenInfo(context.Background(), p.ethClient, paymentToken)
    budgetUSD, _ := GetHistoricalUSDBudget(context.Background(), p.chainID, paymentToken, ev.Amount, tokenInfo.Decimals, e.BlockTimestamp)
    budget := bigIntToFloat(ev.Amount)

    // 写入快照缓存供后续事件复用
    SetJobBudgetCache(ev.JobId.Uint64(), &jobBudgetEntry{
        PaymentToken: paymentToken,
        TokenSymbol:  tokenInfo.Symbol,
        Budget:       ev.Amount,
        BudgetUSD:    budgetUSD,
    })

    job, jobErr := p.getJobWithRetry(ev.JobId.Uint64(), 3)
    // ... 写 commerce_actions ...
}
```

### 4.3 后续事件统一填充 token 字段

**所有写入 `commerce_actions` 的位置**（funded/submitted/终态）：

```go
func (p *CommerceProcessor) fillTokenFields(jobID uint64) (string, string, float64, float64) {
    if cached, ok := GetJobBudgetFromCache(jobID); ok {
        return cached.PaymentToken, cached.TokenSymbol,
               bigIntToFloat(cached.Budget), cached.BudgetUSD
    }
    // 缓存 miss（事件乱序或 indexer 重启）：回退链上查询
    job, err := p.getJobWithRetry(jobID, 3)
    if err != nil {
        return "", "???", 0, 0
    }
    paymentToken := job.PaymentToken.String()
    tokenInfo, _ := ResolveTokenInfo(context.Background(), p.ethClient, paymentToken)
    // Cache miss 时无法获取 BudgetSet 时刻的 timestamp，用当前价格 fallback
    budgetUSD, _ := GetCurrentUSDBudget(context.Background(), p.chainID, paymentToken, job.Budget, tokenInfo.Decimals)
    return paymentToken, tokenInfo.Symbol, bigIntToFloat(job.Budget), budgetUSD
}

// GetCurrentUSDBudget: 用当前价格计算 USD 价值（用于 cache miss 场景）
func GetCurrentUSDBudget(ctx context.Context, platform, tokenAddr string, amount *big.Int, decimals uint8) (float64, error) {
    price, err := fetchPriceWithFallback(ctx, platform, tokenAddr)
    if err != nil {
        log.Warnf("fetchCurrentPrice failed for %s, using default 1: %v", tokenAddr, err)
        price = 1.0
    }
    return float64(amount.Int64()) * price / math.Pow10(int(decimals)), nil
}
```

### 7.2 Token Address → CoinGecko ID 映射

**静态映射（优先）**：
```go
var staticTokenToGeckoID = map[string]string{
    "0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2": "weth",
    "0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48": "usd-coin",
    "0xdAC17F958D2ee523a2206206994597C13D831ec7": "tether",
    "0xEeeeeEeeeEeEeeEeEeEeeEEEeeeeEeeeeeeeEEeE": "ethereum",
}

// resolveGeckoID: 静态表优先，miss 则动态查询
func resolveGeckoID(ctx context.Context, platform, tokenAddr string) (string, error) {
    // 1. 静态映射
    if id, ok := staticTokenToGeckoID[strings.ToLower(tokenAddr)]; ok {
        return id, nil
    }

    // 2. 动态查询: GET /coins/{id}/contract/{contract_address}
    id, err := fetchGeckoIDByContract(ctx, platform, tokenAddr)
    if err != nil {
        return "", err
    }

    // 3. 缓存到静态表（可选，下次直接命中）
    staticTokenToGeckoID[strings.ToLower(tokenAddr)] = id
    return id, nil
}

// fetchGeckoIDByContract: 根据合约地址查 CoinGecko coin ID
func fetchGeckoIDByContract(ctx context.Context, platform, tokenAddr string) (string, error) {
    url := fmt.Sprintf(
        "https://pro-api.coingecko.com/api/v3/coins/%s/contract/%s",
        platform, strings.ToLower(tokenAddr),
    )
    req, _ := http.NewRequest("GET", url, nil)
    req.Header.Set("x-cg-pro-api-key", coingeckoAPIKey)

    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return "", err
    }
    defer resp.Body.Close()

    var result struct {
        ID string `json:"id"`
    }
    json.NewDecoder(resp.Body).Decode(&result)
    if result.ID == "" {
        return "", fmt.Errorf("coin ID not found for %s", tokenAddr)
    }
    return result.ID, nil
}
```

// fetchPriceWithFallback: 多级降级获取 USD 价格
func fetchPriceWithFallback(ctx context.Context, platform, tokenAddr string) (float64, error) {
    // Level 1: Pro API
    if coingeckoAPIKey != "" {
        price, err := fetchPriceFromProAPI(platform, tokenAddr)
        if err == nil {
            return price, nil
        }
        log.Warnf("Pro API failed, fallback to free: %v", err)
    }

    // Level 2: Free API
    price, err := fetchPriceFromFreeAPI(platform, tokenAddr)
    if err == nil {
        return price, nil
    }
    log.Warnf("Free API failed, fallback to default 1: %v", err)

    // Level 3: 默认值 1（适用于稳定币或无法获取价格的情况）
    return 1.0, nil
}

## 八、验收标准

1. `BudgetSet` 事件处理后，`commerce_actions.payment_token` / `token_symbol` / `budget_usd` 有正确值
2. 相同 job 的 `funded`/`submitted`/`completed` 事件复用的 `budget_usd` 与 `BudgetSet` 时刻一致
3. 跨 token（USDC + ETH）job 混合后，`weighted_score_usd` 计算合理（等值 USD 等权）
4. Symbol 缓存命中时无额外 RPC 调用
5. Price 缓存命中时无额外 API 调用
6. `go build ./...` 编译通过
